	\section{Basics of Representation theory}
	
	\begin{definition}[Group representation]
		A \textit{representation} of a group $G$ is a morphism $\rho:G\to \GL(V)$, where $V$ is a certain vector space. Commonly one refers to $V$ as being the representation of $G$, taking as granted from the context the mosphism $\rho$. 
	\end{definition}
	Analogously a representation of a ring or a field is a morphism from it to an arbitrary $\GL(V)$. As it's usual with actions, if $\rho$ is clear from the context we write
	$$\rho_V(g)v=g\circ v=gv$$
	with a slight abuse o notation.
	\begin{definition}[Group algebra]
		Given a field $k$ and a group $G$ the \textit{group algebra} of $G$ is the set
		$$kG:=\left\{\sum_{g\in G}\lambda_g g:\lambda_g\in k\right\}$$
		equipped with the obvious sum (as a vector space) and the distributive multiplication that satisfies the multiplication of $G$.
	\end{definition}
	We should prove that the one above is actually an algebra, but it is quite easy indeed. Given a representation $\rho$ of $G$ we can extend it linearly to a representation $\tilde{\rho}$ such that
	$$\tilde{\rho}:kG \to \text{End}(V)$$
	and this is a ring representation (we go in End because multiplication in $kG$ is not necessarily invertible).
	\begin{definition}[Algebra]
		Given a field $k$, an \textit{algebra} over it is a $k$-vector space equipped with a bilinear, unitary and associative product.
	\end{definition}
	Just as with groups we define a representation of an algebra as a morphism $\rho:A \to \text{End}(V)$.
	\begin{example}[Trivial representation]
		For any group the map
		$\rho:G\to \GL(V)$
		such that
		$$g\mapsto 1$$
		is the so called \textit{trivial representation} of $G$.
	\end{example}
	\begin{definition}[Regular representation]
		For any group $G$, if we take $V=kG$ we get the so called \textit{regular (left) representation} of $G$ as 
		$$h\circ \sum_{g\in G}\lambda_g g \mapsto \sum_{g\in G} h\lambda_g g$$
		this is equivalent to postulate that $G$ acts on $kG$ as the standard left multiplication $h\circ g:= hg$ on elements of the form $g\in kG$, and then on all the other elements we extend it linearly.
	\end{definition}
	\begin{example}[Representation of a permutation]
		For $G=S_n$ and $V=k^n$ the map that permutes the canonical basis is a representation, in other words
		$$g\circ e_i := e_{g(i)}$$
		is a representation.
	\end{example}
	\begin{example}[Determinant]
		Given a representation $\rho:G\to \GL(V)$, with $V$ a vector space on $k$, the map 
		$$\det\circ \rho:G\to \GL(V)\to k$$
		is also a representation (in particular is always $1$-dimensional).
	\end{example}
	
	\subsection{Morphisms of representations}
	
	\begin{definition}[Morphisms of representations]
		Given a group $G$ and two representations $\rho_V$ and $\rho_W$ over the same field $k$, a \textit{morphism} between these is a linear map $T:V\to W$ such that is compatible with both $\rho_V$ and $\rho_W$; that is, we have the following commutative diagram
		\begin{center}
			\begin{tikzcd}
				V\arrow[r,"T"] \arrow[d,"\rho_V(g)",swap]& W\arrow[d,"\rho_W(g)"] \\
				V\arrow[r,"T",swap] & W
			\end{tikzcd}
		\end{center}
		for any $g\in G$. Equivalently we must have
		$$\rho_W(g)\circ T = T \circ \rho_V(g).$$
		These are also called \textit{intertwined operators} or $G$-\textit{equivariant maps}, or $G$-\textit{morphisms}... they have a lot of names. Anyway we say that a morphism of representations is an \textit{isomorphism} if it is invertible.
	\end{definition}
	What we are saying essentially is, being that $\rho_V(g)$ and $\rho_W(g)$ are matrices, $T$ commutes with all such matrices in the way given above. The set of all intertwined operators is oftend denoted as
	$$\text{Hom}(V,W)\ge \text{Hom}_G(V,W)$$
	which is always a non empty vector space (!). \\
	Let's now consider two representations $V$ and $W$, we can investigate what happens to the action of $G$ when we take $V\oplus W$, as we expected
	$$g\circ (v,w)=\left(g\circ v,g\circ w\right)$$
	and no one is surprised. As usual with vector spaces, given two finite dimensional representations $V,W$, if we choose basis $\mathcal{B}_V$, $\mathcal{B}_W$ theese induce a morphism of representations from $V$ and $W$ to
	$$\rho_V:G\to \GL_n(k)$$
	and
	$$\rho_W:G\to \GL_m(k)$$
	in this way $\rho_{V\oplus W}(g)$ has also a nice matrix representation
	$$\rho_{V\oplus W}(g)=
	\left(\begin{array}{c|c}
		\rho_V(g) & 0 \\
		\hline 
		0 & \rho_W(g)
	\end{array}\right).$$
	\begin{lemma}[$G$-equivariant is equivalent to $kG$-equivariant]
		Given a map between two representations $V$ and $W$ of $G$, we have that this is $G$-equivariant iff is $kG$ equivariant, given $k$ the field over which $V$ is defined.
	\end{lemma} 
	\begin{proof}
		If $T$ is $G$-equivariant we have that for any $g\in G$ one has
		$$T\rho_V(g)=\rho_W(g)T$$
		and so extending these to $\tilde{\rho}_V$ and $\tilde{\rho}_W$ we have that, given a certain 
		$$\sum_{g\in G}\lambda_gg\in kG$$
		by linearity of $T$ and $\tilde{\rho}$ we have
		$$T\tilde{\rho}_V\left(\sum_{g\in G}\lambda_gg\right)=\tilde{\rho}_WT\left(\sum_{g\in G}\lambda_gg\right)$$
		which means that $T$ is $kG$-equivariant. On the other hand if $T$ is $kG$-equivariant since $G\subseteq kG$ we have that $T$ is also $G$-equivariant, and hence we have our thesis. 
	\end{proof}
	\begin{example}[Dual representation]
		Given a representation $V$ of $G$, it's dual space $V^*$ has naturally a representation structure
		$$\left(\rho_{V^*}(g)f\right)(v)=f\left(\rho_V(g)^{-1}v\right)$$
		where the inverse is taken to make sure that it is an homomorphism.An equivalent formulation comes with the following commutative diagram:
		\begin{center}
			\begin{tikzcd}
				V\arrow[r,"f"]\arrow[d,"\rho_V(g)",swap] & k \\
				V\arrow[ur,"\rho_{V^*}(g)\circ f",swap] &
			\end{tikzcd}
		\end{center}
		a more compact way to write it is $g\circ f = f\circ g^{-1}$. An even more compact way to see this is through matrices, choosing a basis $B$ and it's dual basis $B^*$ of $V^*$ we see that the actions as matrices satisfy
		$$\rho_{V^*}(g)=\left[\rho_V(g)^{-1}\right]^\intercal.$$
	\end{example}
	\begin{definition}[Tensor product of representations]
		Given two representations $V$ and $W$ of a group their \textit{tensor product} is the space 
		$$V\otimes W := \langle \left\{v \otimes w: v\in V,w\in W\right\}\rangle$$
		equipped with the action 
		$$\rho_{V\otimes W}(g)(v\otimes w):=\rho_V(g)v\otimes \rho_W(g)w.$$
	\end{definition}
	\begin{remark}[Canonical identification of tensor products]
		Given spaces $W$ and $V^*$ there is always a canonical bijection
		\begin{align*}
			F: W\otimes V^* & \inj \text{Hom}(V,W) \\
			\sum c_{wf}w\otimes f & \mapsto \sum c_{wf} f(v)w
		\end{align*}
		which is equivalent to say that on pure tensors our map is just 
		$$w\otimes f \mapsto F(w\otimes f)(v):=f(v)w$$
		and for all the other elements of $V\otimes W$ we extend it linearly. This is just a basis free and abstract way of saying that, given elements $\mathbf{a},\mathbf{b},\mathbf{c}\in\RR^n$ we always have
		$$\left(\mathbf{a}\otimes \mathbf{b}\right)\mathbf{c}:=\left(\mathbf{a}\mathbf{b}^\intercal\right) \mathbf{c}=\mathbf{a}\left(\mathbf{b}^\intercal \mathbf{c}\right)=:\mathbf{a}\langle \mathbf{b},\mathbf{c}\rangle$$  
	\end{remark}
	This gives us immediately that if $V$ and $W$ representations of $G$, so is $V^*\otimes W$. Since the canonical injection defined above is equivariant (prove it) we hence have that \textbf{$\text{Hom}(V,W)$ is a representation of $G$}. What is the action of this? We have that
	$$\left(g\circ f\right)(v)=\rho_W(g)\left(f\left(\rho_V(g^{-1})v\right)\right)=\left(\rho_W(g)\circ f \circ \rho_V(g)^{-1}\right)(v)$$
	and so if $V=W$ this is just a conjugation of $f\in \text{End}(V)$!
	\begin{remark}[$\bigstar$:Internal Automorphisms]
		In a way, if we see a representation of $G$ just as a $H\le \GL(V)\subseteq \text{End}(V)$ monomorphic to $G$, it is natural to think of the action of $G$ on $\text{Hom})(V,V)=\text{End}(V)$ for any $h\in H$ and $f\in \text{End}(V)$ as the conjugation:
		$$h\circ f=hfh^{-1}$$
		in other ways this is just an internal automorphism generated by the elements of the subgroup $H$ of $\GL(V)$.
	\end{remark}
	\subsection{Subrepresentations and Maschke}
	
	\begin{definition}[Subrepresentation]
		Given a representation $V$ of an object $G$, a \textit{subrepresentation} of $V$ is a subspace $U\le V$ such that is $G$-stable, that is 
		$$G\circ U \subseteq U$$
		which means that acting with $G$ on $U$ always gets us to elements still in $U$.
	\end{definition}
	\begin{example}[Fixed point of permutations]
		Given $G=S_n$ with it's permutation representation on $V=k^n$, a subrepresentation of $V$ is
		$$U=\langle e_1+e_2+\dots+e_n\rangle$$
		and in this case is much more than $G$ invariant: is pointwise fixed by the action of $G$. A non pointwise fixed example is the space $W$ with Cartesian equation
		$$\sum_{i=1}^{n}x_i=0$$
		which, if $\text{car}K \not \mid n$ is not pointwise fixed for every point, $w\not\in G\circ w$ for any $w$. 
	\end{example}
	An operation we haven't yet touched is the quotient of spaces, we want to investigate what happens if we take two representation $V\ge U$ of $G$ and consider $\faktor{V}{U}$. We claim that this is a representation with the action
	$$\rho_{V/U}(v+U):=\rho_V(g)v+U$$
	and we should check that this is actually well defined and is a homomorphism. What does this mean in matrix terms? Given a basis $u_i$ of $U$ and a completion of it to a basis of $V$, the matrix of $\rho_V(g)$ in this basis has generally the form
	$$\rho_V(g)=
	\left(
	\begin{array}{c|c}
		\rho_U(g) & * \\
		\hline
		0 & \rho_{V/U}(g)
	\end{array}
	\right)
	$$
	one of the big questions of representation theory, as always when we deal with matrices, is wether, given a certain representation $V$ of a group $G$, we can express the action of $G$ on $V$ as acting through block diagonal matrices. From the formula above we see that if we can find a couple of spaces $U\oplus U'$ that are both $G$ stable, we have achieved our goal. 
	\begin{example}[Decomposition of a representation of $V$]
		Taking $G=S_n$ and $V=k^n$, since we have 
		$$\langle e_1+e_2+\dots+e_n \rangle \oplus \left\langle \left\{\sum_{i=1}^n x_i=0:x\in k^n\right\}\right\rangle$$
		and these are $G$-stable we have that this is a decomposition of $V$ as a representation.
	\end{example}
	\begin{remark}[Lattice of subrepresentations]
		If $V$ is a representation of $G$ and $U,W\le V$ are subrepresentations of $G$ then $U\cap W$ and $U+W$ are subrepresentations of $G$ too.  
	\end{remark}
	\begin{definition}[Fixed points]
		For every representation $V$ of $G$ we can define the set of \textit{fixed points} of $V$ as 
		$$V^G:=\left\{v\in V:gv=v\quad \forall g\in G\right\}$$
		which can also be seen as the sum of all the subrepresentations isomorphic to the trivial one
		$$V^G=\sum_{G\circ u = \left\{u\right\}}\langle u \rangle$$
		this implies that $V^G$ has naturally a subrepresentation structure.
	\end{definition}
	\begin{lemma}[Fixed points of the Hom representation]
		For any pair $V,W$ of representations of $G$ we have that
		$$\text{Hom}_G(V,W)=\text{Hom}(V,W)^G.$$
		seeing $\text{Hom}(V,W)$ as a representation of $G$ through the tensor product.
	\end{lemma}
	\begin{proof}
		From the definition of $\text{Hom}(V,W)^G$ we have
		$$\text{Hom}_G(V,W)=\left\{f\in \text{Hom}(V,W):\rho_V(g)\circ f \circ \rho_W(g)^{-1}=f\right\}$$
		but since 
		$$\rho_V(g)\circ f \circ \rho_W(g)^{-1}=f \iff \rho_V(g)\circ f =f\circ \rho_W(g)$$
		that's just the definition of $G$ equivariant maps, and so we have the thesis.
	\end{proof}
	\begin{definition}[Projector]
		Given a finite group $G$ and a field $k$ we define the \textit{projector} of $G$ in $kG$ as  
		$$\tilde{P}:=\sum_{g\in G}g\in kG$$
		this is always an element of $Z(kG)$. Moreover if $\charin\,k\nmid \,G$ we can define 
		$$P:=\frac{1}{\abs{G}}\tilde{P}$$
		which has the nice property that $P^n=P$ for any $n\in\NN$.
	\end{definition}
	\begin{lemma}[Projection to $V^G$]
		Let $V$ be a representation of a group algebra $G$, then for any $v\in V$ we have
		$$\tilde{\rho}_V(P)v\in V^G$$
		where $\tilde{\rho}_V$ is the regular representation on a group algebra $kG$.
	\end{lemma}
	\begin{proof}
		It follows form the fact that $gP=P=Pg$ for any $g\in G$ that
		$$g\circ\left( P \circ v\right) = gP\circ v = P\circ v$$
		and so this satisfies the definition of $V^G$.
	\end{proof}
	This implies that $\tilde{\rho}_V(P):V\to V^G$ is exactly a projection on $V^G$.
	\begin{lemma}[Kernel and Image of equivariant maps]
		For any $G$-equivariant maps $T:V\to W$ its kernel and image are always $G$-stable.
	\end{lemma}
	\begin{proof}
		For the kernel, given a $g\in G$ and $v\in \ker{T}$ we wish to prove that 
		$$g\circ v\in \ker{T}$$
		on the other hand, since $T$ is equivariant we have
		$$T(g\circ v)=g\circ T(v)=g\circ 0 = 0$$
		which is exactly what we wanted. On the other hand if $w\in \Immagine{T}$ we have that exists a $v\in V$ for which we have
		$$Tv=w$$
		but hence for any $g\in G$ we get
		$$g\circ Tv = T\left(g\circ v\right)\in \Immagine{T}$$
		and we have that also $\Immagine{T}$ is $G$-stable.
	\end{proof}
	this lemma tells us once more that $V^G$ is a subrepresentation of $V$.
	\begin{definition}[Irreducible representation]
		A representation $V$ of $G$ is \textit{irreducible} if its only subrepresentations are itself and the trivial one.
	\end{definition}
	\begin{definition}[Completely reducible]
		A representation $V$ of $G$ is said to be \textit{completely reducible}, or \textit{semisimple} if it can be written as a direct sum of irreducible representations.
	\end{definition}
	\begin{example}[$1$-dimensional representations]
		Every $1$-dimensional representation is irreducible, clearly.
	\end{example}
	\begin{example}[A non irreducible representation]
		Take $C_3$ with the field $\RR$, then its regular representation $\RR C_3$ is not irreducible, since we have the subrepresentation
		$$U=\left\{a1+bg+cg^2:a+b+c=0\right\}.$$
	\end{example}
	We try and think to finite dimensional representations which are not completely reducible, but not many pop up to our minds. This makes us wonder if they even exists, indeed for finite groups we have the strong fact result that is Maschke's Theorem:
	
	\begin{theorem}[Maschke]
		Let $G$ be a finite group such that $\charin k\nmid \abs{G}$. Let $V$ be a representation of $G$. Then any subrepresentation of $G$ has a $G$-stable complement, so as a consequence every finite dimensional representation of $G$ is completely reducible.
	\end{theorem}
	\begin{proof}
		Let $U\le V$ be $G$-stable. If $W'$ is such that
		$$V=U\oplus W'$$
		let $\pi^{W'}_U=:\pi\in \text{Hom}(V,U)$ be the projection onto $U$ along $W'$. We know that $G$ acts on $\text{Hom}(V,U)$ let's take the projector $P$ of $G$, then we have that
		$$\text{Hom}(V,U)^G=\text{Hom}_G(V,U)$$
		and so composing $P\circ \pi$ we get a $G$-equivariant map. By definition of $P$ we have
		$$P\circ \pi = \frac{1}{\abs{G}}\sum_{g\in G} g\circ \pi=\frac{1}{\abs{G}}\sum_{g\in G}\rho_U(g)\circ \pi\circ \rho_V(g)^{-1}$$
		where at the RHS the $\circ$ represents function composition and on the LHS represents action. How does this function behave on $U$? If $u\in U$ we have
		$$(P\circ \pi) u=\frac{1}{\abs{G}}\sum_{g\in G}\rho_U(g)\circ \pi\circ \rho_V(g)^{-1}(u)=\frac{1}{\abs{G}}\sum_{g\in G}g^{-1}\circ g\circ u=u$$
		so $P\circ \pi_{\mid U}=\text{id}_{\mid U}$ but then since $P\circ \pi$ is equivariant (is the composition of equivariant maps), it's kernel must be $G$-stable, since the kernel is just $W'$, we are done.
	\end{proof}
	\begin{corollary}[Strong Maschke]
		For any representation $V$ of $G$ we have that the following are equivalent:
		\begin{itemize}
			\item Every subrepresentation of $V$ has a stable complement.
			\item For any representation $V$ we have irreducible subrepresentations $V_i$ such that
			$$V=\bigoplus_{i=1}^k V_i.$$
			\item For any representation $V$ we have irreducible subrepresentations $V_i$ such that
			$$V:=\sum_{i=1}^k V_i.$$
		\end{itemize}
		In particular Maschke's theorem tells us that the first condition is true, and so all the other ones must follow as well.
	\end{corollary}
	
	\subsubsection{Schur's Lemma}
	
	Another strong result about irreducible representations is Schur's lemma:
	\begin{lemma}[Schur]
		For any $G$-equivariant map $\phi$, which is not identically zero, between representations $V$ and $W$ of $G$ we have the following:
		\begin{itemize}
			\item[(i)] If $V$ is irreducible then $\phi$ is injective.
			\item[(ii)] If $W$ is irreducible then $\phi$ is surjective.
		\end{itemize}
		Thus if both $V$ and $W$ are irreducible we have that $\phi$ must be an isomorphism, and hence the space of isomorphism between any two irreducible representations $\text{Hom}_G(V,W)$ is either the null set (the set containing just the null element) or an algebra where every element can be inverted!
	\end{lemma}
	\begin{proof}
		In the first case we have that 
		$$\ker{\phi}\le V$$
		is a subrepresentation of $V$, but since $\phi$ is not identically zero we must have that $\ker{\phi}=\left\{0\right\}$, and hence $\phi$ is injective. On the other hand in the second case we have 
		$$\Immagine{\phi}\le W$$
		is a subrepresentation of $W$, and since $\phi$ is not identically zero we must have $\Immagine{\phi}=W$ which means that $\phi$ is surjective.
	\end{proof}
	In the case of algebraically closed fields this result is much much stronger:
	\begin{corollary}[Schur's lemma for algebraically closed fields]
		Le $V,W$ be irreducible representations of $G$, if $k$ is algebraically closed then $\text{End}_G(V)=k\text{id}_V$.
	\end{corollary}
	\begin{proof}
		\footnote{This proof works only for finite dimension, although the result works for general representations as well.}
		For the first part we simply have that iff there exists a nonzero $f\in \text{Hom}_G(V,W)$ then we have $\ker{f}\le V$ is a subrepresentation of $V$, which means that $\ker{f}=0$. Thus we have $f$ injective, which means that 
		$$V\cong \Immagine{f}\le W$$
		which by irreducibility implies $V\cong W$. For the second part we have that, given a $f\in \text{End}_G(V)$ for the algebraic closeness of $k$ we always have an eigenvalue of $f$, let's call it $\lambda$. But this means that
		$$0\ne \ker{f-\lambda I_V}$$
		is $G$-stable, so it must be $V$.
	\end{proof}
	\begin{corollary}[Triviality of action on irreducible representations]
		If $k$ is algebraically closed then any $z\in Z(kG)$ acts as a scalar on any irreducible representation.
	\end{corollary}
	
	\subsection{Group functionals and Characters}
	\begin{definition}[Class function]
		For any group $G$ and field $k$ a \textit{group functional} from $G$ to $k$ is just a function $f:G\to k$, the set of all such functions is denoted with $F(G,k)$. The set of group functionals which are constant over the conjugacy classes of $G$ is denoted as $F_c(G,k)$.
	\end{definition}

	\begin{definition}[Character]
		Let $V$ be a representation of $G$, then the \textit{character} of $V$ is
		\begin{align*}
			\chi_V:G & \to k \\
			g & \mapsto \tr{\rho_V(g)}

		$$\rho_W(g)=T\rho_V(g)T^{-1}$$
		and so $\chi_V(g)=\chi_W(g)$ for any $g\in G$. This means that on any pair of isomorphic representations, the character is the exact same function!
	\end{remark}

	\begin{definition}[Inner product on group functionals]
		On $F(G,\CC)$ and $F_C(G,\CC)$ we define an inner product

	\end{definition}
	\begin{proposition}[First Orthogonality relation]
		If $k=\CC$ and $\abs{G}<\infty$, given two representations $V,W$ we have that
		$$\left(\chi_V,\chi_W\right)_G=\dim\text{Hom}_G(V,W)=\overline{\dim \text{Hom}_G(V,W)}$$
		and because that number is an integer, in particular is purely real, 
	\end{proposition}
	\begin{proof}
		We want to compute 
		$$\left(\chi_V,\chi_W\right)_G=\frac{1}{\abs{G}}\sum_{g\in G}\chi_V(g)\overline{\chi_W(g)}$$
		since we are on $\CC$ the character of the dual representation is just the conjugate, and so we have
		$$\dots=\frac{1}{\abs{G}}\sum_{g\in G}\chi_{V\otimes V^*}(g)$$
		so through linearity we get
		$$\dot= \chi_{V\otimes W^*}\left(\frac{1}{\abs{G}}\sum_{g\in G}g\right)=\chi_{V\otimes W^*}(P)=\tr{P_{\mid V\otimes W^*}}$$
		but the trace of a projection is just the dimension of the image, so since $P$ is a projection from $\text{Hom}(V,W)$ to $\text{Hom}_G(V,W)$ we have our result. 
	\end{proof}
	So putting together the result of Maschke and orthogonality of characthers we get that the set $\text{Irr}(G)$ of irreducible representations of $G$ are always
	$$\abs{\text{Irr}(G)}\le \dim F_C(G,\CC)$$
	and so we have that any $\chi_V$ is a $\ZZ_{\ge 0}$-linear combination of $\chi_{V_i}$. 
	\begin{corollary}[Fourier Coefficients of the characters]
		If $\left\{V_i,i\in I\right\}$ is a set of isomorphsim classes of irreducible representations of $G$ we have that any $V$ representation of $G$ is
		$$V=\bigoplus_{i=1}^k V_i^{\oplus n_i}$$
		moreover we have 
		$$\left(\chi_V,\chi_i\right)_G=n_i$$
		so the $n_i$ are the fourier coefficients of $\chi_V$ in the orthonormal basis $\chi_i$. This means in particular that the multiplicities $n_i$ are unique.
	\end{corollary}
	
	\begin{proposition}[Equality of Characters implies equality of representation]
		Let $(V,\rho)$ and $(W,\rho')$ be two representations of a group $G$ over complex numbers. If $\chi_\rho = \chi_{\rho'}$ then $V=W$.
	\end{proposition}
	\begin{proof}
		Let's consider the set of all irreducible representations of $G$ as
		$$\text{Irr}_G(G)=\left\{V_1,\dots V_r\right\}$$
		now by Maschke's theorem we can decompose both $W$ and $V$ as a sum of irreducibles:
		\begin{gather*}
			V=\bigoplus_{j=1}^rV_i^{\oplus n_i} \\
			W=\bigoplus_{j=1}^rV_i^{\oplus m_i}
		\end{gather*}
		since the character of a sum of representations is the sum of the character, denoted with $\chi_i$ the characters of $V_i$ we have that
		$$\chi_\rho = \sum_{i=1}^r n_i\chi_i$$
		but since the character are the same we have that
		$$\sum_{i=1}^r n_i\chi_i=\sum_{i=1}^r m_i\chi_i$$
		and so taking the inner product on both sides wrt to any $\chi_i$ we have
		$$n_i=\left(\chi_\rho,\chi_i\right)=\left(\chi_{\rho'},\chi_i\right)=m_i$$
		and hence $V=W$ as desired.
	\end{proof}
	\begin{corollary}[Characterisation of irreducibility]
		If $V$ is a finite dimensional representation of $G$ on $\CC$ then $V$ is irreducible if and only if 
		$$(\chi_V,\chi_V)=1.$$
	\end{corollary}
	\begin{proof}
		We decompose $V$ as a sum of irreducible representations again
		$$V=\bigoplus_{i=1}^r V_i^{\oplus n_i}$$
		from this we have a very explicit formula for $\chi_V$, which we can use to compute the quantity we are considering
		$$\left(\chi_V,\chi_V\right)=\left(\sum_{i=1}^r \chi_in_i,\sum_{i=1}^r \chi_in_i\right).$$
		If that expression is $1$ we see immediately that we must have $n_j=\delta_{ij}$ for some $i\in \NN$, and hence $V$ is irreducible. The other side of the thesis follows immediately from first orthogonality relations, and so we are done.
	\end{proof}
	\begin{corollary}[$V$ is irreducible iff $V^*$ is irreducible]
		Let $V$ be a finite dimensional complex representation of a finite group $G$ then $V$ is irreducible if and only id $V^*$ is irreducible.
	\end{corollary}
	\begin{proof}
		We have the following chain of equalities
		$$1=\left(\chi_V,\chi_V\right)=\overline{\left(\chi_V,\chi_V\right)}=\left(\overline{\chi_V},\overline{\chi_V}\right)=\left(\chi_{V^*},\chi_{V^*}\right)$$
		hence we have proven that $V^*$ is irreducible by the previous result. The converse follows from identifying $V^{**}$ with $V$ invoking the finite dimensionality of $V$.
	\end{proof}
	\begin{lemma}[Action of the regular representation]
		For any finite $G$ acting on $\CC G$ through the regular representation we have that
		$$\chi_{\text{reg}}(g)=\abs{G}\delta_{e g}$$
		where $e$ is the identity.
	\end{lemma}
	\begin{proof}
		It's clear that the character if $g=e$ the matrix we are considering is the identity, and so we have our thesis. \\
		The difficult case is when $g\ne e$. In the action of $g$ on $\CC G$ we see that 
		$$e_h \mapsto e_{gh}$$
		and so in the basis $\left\{e_g:g\in G\right\}$ every column of the matrix of $\rho(g)$ consists of just a non zero entry, equal to $1$. Moreover we can't have that if $h_1\ne h_2$
		$$e_{gh_1}=e_{gh_2}$$ 
		since acting left and right by $g^{-1}$ we have $h_1=h_2$. This mean that $\rho(g)$ is just a permutation matrix, and since $g\ne e$ we have
		$$e_{gh}\ne e_h$$
		for any $h$, this means that along the main diagonal there are only zeros, and hence the theorem follows. 
	\end{proof}
	\begin{theorem}[Decomposition of the regular representation]
		Let $V_1, \dots, V_k$ be the complete set of irreducible representations of $G$ and let $\chi_i$ be the corresponding characters. Then the regular representation decomposes as 
		$$\CC G = \bigoplus_{i=1}^k V_i^{\oplus \dim V_i}.$$
	\end{theorem}
	\begin{proof}
		Let's calculate the fourier coefficients of $\chi_{reg}$ using the previous lemma:
		$$\left(\chi_{i},\chi_{reg}\right)=\frac{1}{\abs{G}}\sum_{g\in G}\chi_i(g)\overline{\chi_{reg}(g)}=\frac{1}{\abs{G}}\abs{G}\delta_{e g}\chi_i(e)=\dim V_i$$
		which by uniqueness of the decomposition in irreducible is our thesis. Explicitly we have that if 
		$$V=\bigoplus V_i^{\oplus n_i}$$
		then computing the inner product using this formula and the additivity of characters we gain
		$$\dim V_i=\left(\chi_i,\chi_{reg}\right)=\left(\chi_i,\sum n_i\chi_i\right)=n_i$$
		and hence we are done.
	\end{proof}
	\begin{corollary}[Sum of squares formula]
		For any group $G$, given the set of irreducible representations $V_i$ we always have that
		$$\abs{G}=\sum_{i=1}^k \left(\dim V_i\right)^2.$$
	\end{corollary}
	\begin{proof}
		Taking the dimension on both sides of the formula 
		$$\CC G = \bigoplus_{i=1}^k V_i^{\oplus \dim V_i}$$
		we have the thesis.
	\end{proof}
	
	\subsection{Commutators and abelian groups}
	
	Let's consider a finite group $G$ of order $d$. Given a representation
	$$\rho : G \to \GL(V)$$
	we have that for every $g\in G$ this is a matrix. Moreover by Lagrange's theorem we have that
	$$g^d=1 \implies \rho(g)^d = \text{id}$$
	this means that denoted with $X:= \rho(g)$ its minimal polynomial $m_X$ satisfies
	$$m_X \mid x^d-1$$
	now we make two important assumption about our field:
	\begin{enumerate}
		\item All the roots of $x^d-1$ exist in $k$.
		\item We have $(\charin k, d)=1$ (this is to make sure that we have no repeated roots in $x^d-1$).
	\end{enumerate}
	Then we have that $m_X$ factors as a product of distinct linear factors, and we have diagonalizability! All the above assumptions are naturally satisfied if we work on $\CC$, which is anyway the field where we'll be working on. We need here an ounce of linear algebra
	\begin{proposition}[Simultaneous diagonalizability]
		Let $X,Y\in \text{End}(V)$ diagonalizable. Then $X$ and $Y$ are simultaneously diagonalizable iff they commute.
	\end{proposition}
	\begin{proof}
		If they are simultaneously diagonalizable the thesis clearly holds, since all diagonal matrices commute. To prove the converse we decompose $V$ in $X$ eigenspaces
		$$V=\bigoplus E_\lambda(X)$$
		now if we apply $Y$ to any $E_\lambda(X)$ we stay in $E_\lambda(X)$, in fact if $v\in E_\lambda(X)$ we have through commutativity
		$$X(Yv)=YXv=Y\lambda v= \lambda Yv$$
		which just implies $Yv\in E_\lambda(X)$. But this immediately concludes, because 
		$$X_{\mid E_\lambda(X)}=\lambda\text{id}$$
		and so since $Y$ is again diagonalizable in $E_\lambda(X)$ if we just pick any basis of $E_\lambda(X)$ that diagonalizes $Y$ we get for free that $X$ is diagonal in that same base (since it is diagonal in any base!). 
	\end{proof}
	Putting together theese two results we have that if $A$ is a finite abelian group, then for any representation $V$ we can pick a basis of $V$ where all $\rho(a)$ are diagonal matrices! A fun little application is:
	\begin{corollary}[Representation of finite ciclyc groups]
		For any $r\in\NN$ and $d\in \NN$ we have that if
		$$\left(\faktor{\ZZ}{d\ZZ}\right)^r \hookrightarrow \GL_n(\CC)$$
		is an injective morphism then $r\le n$.
	\end{corollary}
	Let $A$ be an abelian finite group and $\rho:A\to \GL(V)$. If each $\rho(g)\in \GL(V)$ is diagonalizzable then all elements $\rho(g)$ are diagonalizable, so if we take 
	$$V_i:=\langle v_i \rangle$$
	where $v_i$ is the basis that diagonalizes all the $\rho(g)$. For any $g\in A$ and $i\le \dim V$ then we have that
	$$\rho(g)V_i \subseteq V_i$$
	which means that for all $i$ the $V_i$s are subrepresentations of $V$. But it's clear that all $\rho(g)$ are diagonalizable, since we saw earlier that they all must satisfy
	$$\left[\rho(g)\right]^{\abs{G}}-1=0$$
	which we have assumed has all distinct roots in $k$.
	From this discussion we evince
	\begin{proposition}[Irreducible representations of finite abelian groups]
		All the irreducible representations of a finite abelian group $A$ are the finite dimensional ones. Moreover from the sum of squares formula we have that there are exactly $\abs{A}$ irreducible representations of $A$.
	\end{proposition}
	From the structure theorem of finite abelian groups we know that any finite abelian $A$ decomposes as
	$$A\cong C_{p_1^{e_1}}\times C_{p_2^{e_2}}\times \dots \times C_{p_k^{e_k}}$$
	where $p_i$ are not necessarily distinct primes. This means we can find $g_1,\dots,g_k$ elements of $A$ (ideally the elements $(a_1,1,\dots,1)$, $(1,a_2,\dots,1)$, $\dots$, $(1,1,\dots,a_k)$, generators of the canonical isomorphic images of $C_{p_i^{e_i}}$ in $A$) of order respectively $p_i^{e_i}$. Then choosing the representation of $A$ amounts to just choosing $\rho(g_i)$ for all $g_i$. For any of these elements we have $p_i^{e_i}$ choices, and since this choices are all independent the number of distinct representations of $A$ are $p_1^{e_1}p_2^{e_2}\dots p_k^{e_k}$, which is exactly what we concluded in the previous result. 
	\begin{definition}[Commutator]
		For any group $G$ the function $G\times G\to G$ such that
		$$[g,h]:=ghg^{-1}h^{-1}$$
		is called the \textit{commutator} of $g$ and $h$. The \textit{derived subgroup} of $G$ (also called the commutator group) is defined to be
		$$[G,G]:=\left\langle[g,h]:g,h\in G\right\rangle$$
		which is a normal subgroup of $G$, the \textit{abelianization} of $G$ is the group $\faktor{G}{[G,G]}$.
	\end{definition}
	We immediately see that for any one dimensional representation $\rho:G\to V$ of any finite group $G$ we have that
	$$\rho([g,h])=1$$
	since the image is commutative. From a generalization of the first homomorphism theorem we evince immediately that
	\begin{center}
		\begin{tikzcd}
			G\arrow[r,"\rho"]\arrow[d,"\pi",swap] & \CC^* \\
			\faktor{G}{[G,G]}\arrow[ur,"\overline{\rho}",swap] & 
		\end{tikzcd} 
	\end{center}
	and so immediately we can decompose any representation of any group $G$ in a product of two morphisms, one that is just a representation of an abelian group, and the other one which is a group morphism.
	\begin{corollary}[Number of monodimensional representations]
		If $k$ is algebraically closed and $(\charin k, \abs{G})=1$ we have that the number of finite dimensional representations of $G$ is $\abs{G:[G:G]}$.
	\end{corollary}
	\begin{proof}
		Follows from the number of $1$-dimensional representations of abelian groups.
	\end{proof}
	\begin{example}[Symmetric one dimensional representations]
		How many one dimensional representations of $S_n$ are there? 
	\end{example}
	\begin{proof}[Solution]
		This boils down to determining the representation of the abelianization of $S_n$. Since for any $g_1,g_2\in S_n$ we have
		$$\text{sign}([g_1,g_2])=\text{sign}(g_1)\text{sign}(g_2)\text{sign}(g_1)^{-1}\text{sign}(g_2)^{-1}=1$$
		this means that
		$$[G,G]\le A_n$$
	\end{proof}
	
	\subsection{Character tables}
	
	Let's return to class functions. For any $f\in F_c(G,\CC)$ and any $\rho:G\to\GL(V)$ we can define an action of $f$ on $V$ in the following way
	$$s_f:=\sum_{g\in G}f(g)\rho(g)\in \text{Hom}_\CC(V,V).$$
	This has some quite nice properties, for example we have that
	$$s_1=\sum_{g\in G}\rho(g)$$
	which is basically the projector multiplied by $\abs{G}$. 
	\begin{remark}
		The projector $P$ has the property that 
		$$\rho(h)(Pv)=Pv$$
		for any $v\in V$, and it really is the projection transformation onto the space of fixed points of $V$, that is $V^G$.
	\end{remark}
	Actually the property of commutation enjoyed by the projector is much more general:
	\begin{lemma}[Commutativity of class functions]
		For any $f\in F_c(G,\CC)$ and $h\in G$ we have that
		$$\rho(h)s_f=s_f\rho(h)$$
		that is, $s_f$ is $G$-equivariant.
	\end{lemma}
	\begin{proof}
		An explicit calculation, we have
		$$\rho(h)\left(\sum_{g\in G}f(g)\rho(g)\right)\rho(h)^{-1}=\sum_{g\in G}f(g)\rho(hgh^{-1})=\sum_{g\in G}f(g)\rho(g)$$
		which proves our thesis.
	\end{proof}
	\begin{proposition}[Trace and inner product formula]
		For any $f\in F_c(G,\CC)$ we have that
		$$\chi_\rho(s_f)=\sum_{g\in G}f(g)\chi_\rho(g).$$
	\end{proposition}
	\begin{proof}
		Follows form linearity of trace.
	\end{proof}
	From this last property follows that for any $f\in F(G,k)$ one has
	$$\abs{G}(f,\overline{\chi}_\rho)=\chi_\rho(s_f).$$
	\begin{theorem}[Orthonormality of characters]
		Let $\chi_1,\chi_2,\dots,\chi_n$ be all the irreducible characters of a group $G$ (that is, the characters associated with any irreducible representation of $G$), then this is an orthonormal base of $F_c(G,\CC)$.
	\end{theorem}
	\begin{proof}
		We already proved that the characters are an orthonormal set, in fact
		$$\left(\chi_i,\chi_j\right)=\delta_{ij}$$
		it remains to be shown that these span all $F_c(G,k)$. Let's consider $f\in F_c(G,\CC)$ such that
		$$f\in H^\perp$$
		where
		$$H=\langle \chi_1,\dots ,\chi_k\rangle$$
		this is equivalent to saying that for all $\chi_i$ we have 
		$$\left(f,\chi_i\right)=0 \implies \left(f,\overline{\chi}_i\right)=0$$
		but we proved before that
		$$(f,\overline{\chi}_i)=\frac{\chi_i(s_f)}{\abs{G}}$$
		now by schur's lemma we now have that (since we are working on a closed field), since $s_f$ is an equivariant map, over any $V_i$ irreducible representation associated with $\chi_i$ we have that $s_f$ acts as a scalar on every irreducible representation of $G$, but since this scalar $k$ is just given by 
		$$\chi_i(s_f)=k\dim_\CC V_i$$
		we get that $s_f$ is identically on any $V_i$, which implies that $f=0$, as desired. 
	\end{proof}
	\begin{corollary}[Conjugacy classes of a group]
		The number of conjugacy classes of $G$ equal the number of it's irreducible representations.
	\end{corollary}
	\begin{proof}
		Since the dimension of $F_c(G,\CC)$ is just the number of conjugacy classes of $G$, we have that $k$, the number of irreducible characters, is equal to the dimension of $F_c(G,k)$.
	\end{proof}
	
	\begin{example}[Dihedral group]
		For a dihedral group $D_4$ we have that the group of commutators is
		$$[D_4,D_4]=\langle r^2\rangle$$
		where $r$ is an element of order $4$. This means that the number of one dimensional representations of $D_4$ is just $4$. From the sum of squares formula we have 
		$$8=1^2+1^2+1^2+1^2+\dots$$
		what we are missing is a four, but since we have just listed all one dimensional representation of $G$ that four must be written as a $2^2$, which means that all the irreducible representations of $D_4$ are just the $4$ one dimensional one and the $2$ dimensional one. This implies that $D_4$ has exactly $5$ characters (or $5$ conjugacy classes).
	\end{example}
